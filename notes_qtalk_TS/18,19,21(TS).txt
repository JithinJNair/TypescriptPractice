class Animal{
     makesound(){
          console.log('Animal sound');
          
     }
}
class Dog extends Animal{
      makesound(): void {
        console.log('bark');
        
      }
}
class Cat extends Dog{
   makesound(): void {
      console.log('memowww');
      
   }
}
let a = new Animal()
a.makesound()

let a2 = new Cat()
